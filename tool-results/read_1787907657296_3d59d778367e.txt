    83→/* --------------------------- record value rendering ------------------------ */
    84→
    85→function renderCellValue(value: unknown): string {
    86→  if (value === null || value === undefined || value === "") return "—";
    87→  if (typeof value === "boolean") return value ? "TRUE" : "FALSE";
    88→  if (Array.isArray(value)) return value.join(", ");
    89→  if (typeof value === "object") return JSON.stringify(value);
    90→  return String(value);
    91→}
    92→
    93→/* -------------------------------- CSV export ------------------------------- */
    94→
    95→function csvEscape(v: unknown): string {
    96→  const s = renderCellValue(v);
    97→  return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
    98→}
    99→
   100→function exportSystemCsv(
   101→  systemName: string,
   102→  cols: BlueprintField[],
   103→  rows: SystemRecordDTO[]
   104→) {
   105→  const headers = cols.length > 0 ? cols.map((c) => c.label) : ["data"];
   106→  const lines = [headers.map((h) => csvEscape(h)).join(",")];
   107→  for (const rec of rows) {
   108→    const data = parseRecordData(rec.data);
   109→    const row =
   110→      cols.length > 0
   111→        ? cols.map((c) => csvEscape(data[c.key]))
   112→        : [csvEscape(JSON.stringify(data))];
   113→    lines.push(row.join(","));
   114→  }
   115→  const blob = new Blob([lines.join("\n")], {
   116→    type: "text/csv;charset=utf-8",
   117→  });
   118→  const url = URL.createObjectURL(blob);
   119→  const a = document.createElement("a");
   120→  a.href = url;
   121→  a.download = `${systemName.toLowerCase().replace(/[^a-z0-9]+/g, "-")}-records.csv`;
   122→  document.body.appendChild(a);
   123→  a.click();
   124→  a.remove();
   125→  URL.revokeObjectURL(url);
   126→}

   130→/** Tiny gold bar chart: records created per day over the last 14 days. */
   131→function RecordsPulse({ records }: { records: SystemRecordDTO[] }) {
   132→  const days = useMemo(() => {
   133→    const buckets = new Map<string, number>();
   134→    for (let i = 13; i >= 0; i--) {
   135→      const d = new Date();
   136→      d.setDate(d.getDate() - i);
   137→      buckets.set(d.toISOString().slice(0, 10), 0);
   138→    }
   139→    for (const rec of records) {
   140→      const key = new Date(rec.createdAt).toISOString().slice(0, 10);
   141→      if (buckets.has(key)) buckets.set(key, (buckets.get(key) ?? 0) + 1);
   142→    }
   143→    return [...buckets.entries()];
   144→  }, [records]);
   145→
   146→  const max = Math.max(1, ...days.map(([, n]) => n));
   147→  return (
   148→    <div
   149→      className="flex h-12 items-end gap-[3px]"
   150→      role="img"
   151→      aria-label="Records created per day, last 14 days"
   152→    >
   153→      {days.map(([day, n]) => (
   154→        <div
   155→          key={day}
   156→          title={`${day} · ${n} record${n === 1 ? "" : "s"}`}
   157→          className="group relative flex-1 rounded-t-[2px] transition-colors"
   158→          style={{ height: `${Math.max(8, (n / max) * 100)}%` }}
   159→        >
   160→          <div
   161→            className={cn(
   162→              "absolute inset-0 rounded-t-[2px] transition-colors",
   163→              n > 0 ? "bg-gold/70 group-hover:bg-gold" : "bg-foreground/10"
   164→            )}
   165→          />
   166→        </div>
   167→      ))}
   168→    </div>
   169→  );
   170→}

   174→function DynamicFieldInput({
   175→  field,
   176→  value,
   177→  onChange,
   178→}: {
   179→  field: BlueprintField;
   180→  value: unknown;
   181→  onChange: (v: unknown) => void;
   182→}) {
   183→  const type = field.type;
   184→  if (type === "boolean" || type === "bool" || type === "checkbox") {
   185→    return (
   186→      <div className="flex h-9 items-center gap-2 rounded-md border border-border bg-secondary/60 px-3">
   187→        <Switch
   188→          checked={!!value}
   189→          onCheckedChange={onChange}
   190→          aria-label={field.label}
   191→        />
   192→        <span className="font-mono text-[11px] text-muted-foreground">
   193→          {value ? "TRUE" : "FALSE"}
   194→        </span>
   195→      </div>
   196→    );
   197→  }
   198→  if ((type === "select" || type === "enum") && field.options?.length) {
   199→    return (
   200→      <Select value={String(value ?? "")} onValueChange={onChange}>
   201→        <SelectTrigger className="h-9 w-full border-border bg-secondary/60">
   202→          <SelectValue placeholder={`Select ${field.label.toLowerCase()}`} />
   203→        </SelectTrigger>
   204→        <SelectContent className="border-border bg-popover">
   205→          {field.options.map((opt) => (
   206→            <SelectItem key={opt} value={opt}>
   207→              {opt}
   208→            </SelectItem>
   209→          ))}
   210→        </SelectContent>
   211→      </Select>
   212→    );
   213→  }
   214→  const htmlType =
   215→    type === "number" || type === "int" || type === "float"
   216→      ? "number"
   217→      : type === "date" || type === "datetime"
   218→        ? "date"
   219→        : "text";
   220→  return (
   221→    <Input
   222→      type={htmlType}
   223→      value={value === null || value === undefined ? "" : String(value)}
   224→      onChange={(e) =>
   225→        onChange(htmlType === "number" ? e.target.value : e.target.value)
   226→      }
   227→      className="h-9 border-border bg-secondary/60 focus-visible:ring-gold/50"
   228→    />
   229→  );
   230→}
