    65→/* --------------------------- record value rendering ------------------------ */
    66→
    67→export function renderCellValue(value: unknown): string {
    68→  if (value === null || value === undefined || value === "") return "—";
    69→  if (typeof value === "boolean") return value ? "TRUE" : "FALSE";
    70→  if (Array.isArray(value)) return value.join(", ");
    71→  if (typeof value === "object") return JSON.stringify(value);
    72→  return String(value);
    73→}
    74→
    75→/* -------------------------------- CSV export ------------------------------- */
    76→
    77→function csvEscape(v: unknown): string {
    78→  const s = renderCellValue(v);
    79→  return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
    80→}
    81→
    82→export function exportSystemCsv(
    83→  systemName: string,
    84→  cols: BlueprintField[],
    85→  rows: SystemRecordDTO[],
    86→  suffix = "records"
    87→) {
    88→  const headers = cols.length > 0 ? cols.map((c) => c.label) : ["data"];
    89→  const lines = [headers.map((h) => csvEscape(h)).join(",")];
    90→  for (const rec of rows) {
    91→    const data = parseRecordData(rec.data);
    92→    const row =
    93→      cols.length > 0
    94→        ? cols.map((c) => csvEscape(data[c.key]))
    95→        : [csvEscape(JSON.stringify(data))];
    96→    lines.push(row.join(","));
    97→  }
    98→  const blob = new Blob([lines.join("\n")], {
    99→    type: "text/csv;charset=utf-8",
   100→  });
   101→  const url = URL.createObjectURL(blob);
   102→  const a = document.createElement("a");
   103→  a.href = url;
   104→  a.download = `${systemName.toLowerCase().replace(/[^a-z0-9]+/g, "-")}-${suffix}.csv`;
   105→  document.body.appendChild(a);
   106→  a.click();
   107→  a.remove();
   108→  URL.revokeObjectURL(url);
   109→}

   113→function DynamicFieldInput({
   114→  field,
   115→  value,
   116→  onChange,
   117→}: {
   118→  field: BlueprintField;
   119→  value: unknown;
   120→  onChange: (v: unknown) => void;
   121→}) {
   122→  const t = useT();
   123→  const type = field.type;
   124→  if (type === "boolean" || type === "bool" || type === "checkbox") {
   125→    return (
   126→      <div className="flex h-9 items-center gap-2 rounded-md border border-border bg-secondary/60 px-3">
   127→        <Switch
   128→          checked={!!value}
   129→          onCheckedChange={onChange}
   130→          aria-label={field.label}
   131→        />
   132→        <span className="font-mono text-[11px] text-muted-foreground">
   133→          {value ? t.recs.true : t.recs.false}
   134→        </span>
   135→      </div>
   136→    );
   137→  }
   138→  if ((type === "select" || type === "enum") && field.options?.length) {
   139→    return (
   140→      <Select value={String(value ?? "")} onValueChange={onChange}>
   141→        <SelectTrigger className="h-9 w-full border-border bg-secondary/60">
   142→          <SelectValue placeholder={t.recs.selectField(field.label)} />
   143→        </SelectTrigger>
   144→        <SelectContent className="border-border bg-popover">
   145→          {field.options.map((opt) => (
   146→            <SelectItem key={opt} value={opt}>
   147→              {opt}
   148→            </SelectItem>
   149→          ))}
   150→        </SelectContent>
   151→      </Select>
   152→    );
   153→  }
   154→  const htmlType =
   155→    type === "number" || type === "int" || type === "float"
   156→      ? "number"
   157→      : type === "date" || type === "datetime"
   158→        ? "date"
   159→        : "text";
   160→  return (
   161→    <Input
   162→      type={htmlType}
   163→      value={value === null || value === undefined ? "" : String(value)}
   164→      onChange={(e) =>
   165→        onChange(htmlType === "number" ? e.target.value : e.target.value)
   166→      }
   167→      className="h-9 border-border bg-secondary/60 focus-visible:ring-gold/50"
   168→    />
   169→  );
   170→}

   174→function SelectBox({
   175→  checked,
   176→  indeterminate,
   177→  onChange,
   178→  disabled,
   179→  label,
   180→}: {
   181→  checked: boolean;
   182→  indeterminate?: boolean;
   183→  onChange: () => void;
   184→  disabled?: boolean;
   185→  label: string;
   186→}) {
   187→  return (
   188→    <button
   189→      type="button"
   190→      role="checkbox"
   191→      aria-checked={indeterminate ? "mixed" : checked}
   192→      aria-label={label}
   193→      disabled={disabled}
   194→      onClick={(e) => {
   195→        e.stopPropagation();
   196→        onChange();
   197→      }}
   198→      className={cn(
   199→        "flex size-4 shrink-0 items-center justify-center rounded-[3px] border transition-all",
   200→        "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold/60 focus-visible:ring-offset-1 focus-visible:ring-offset-card",
   201→        checked || indeterminate
   202→          ? "border-gold bg-gold text-black hover:bg-gold-pale"
   203→          : "border-border bg-secondary/50 hover:border-gold/60 hover:bg-secondary",
   204→        disabled && "cursor-not-allowed opacity-40"
   205→      )}
   206→    >
   207→      {indeterminate ? (
   208→        <span className="h-0.5 w-2 rounded-full bg-black" />
   209→      ) : checked ? (
   210→        <Check className="size-3" strokeWidth={3} />
   211→      ) : null}
   212→    </button>
   213→  );
   214→}

   226→/* ------------------------------- records tab ------------------------------- */
   227→
   228→export function RecordsTab({
   229→  systemId,
   230→  systemName,
   231→  isArchived,
   232→  records,
   233→  columns,
   234→  invalidateAll,
   235→}: {
   236→  systemId: string;
   237→  systemName: string;
   238→  isArchived: boolean;
   239→  records: SystemRecordDTO[];
   240→  columns: BlueprintField[];
   241→  invalidateAll: () => void;
   242→}) {
   243→  const t = useT();
   244→  const [addingRecord, setAddingRecord] = useState(false);
   245→  const [draft, setDraft] = useState<Record<string, unknown>>({});
   246→  const [editingId, setEditingId] = useState<string | null>(null);
   247→  const [editDraft, setEditDraft] = useState<Record<string, unknown>>({});
   248→  const [selected, setSelected] = useState<Set<string>>(new Set());
   249→  const [confirmBulk, setConfirmBulk] = useState(false);
   250→  /** keyboard cursor — index into `records`, moved with j/k, armed with x/e */
   251→  const [cursor, setCursor] = useState<number | null>(null);
   252→  const rowRefs = useRef<Map<number, HTMLTableRowElement>>(new Map());
   253→
   254→  // keep the cursor valid as records come and go (adjust-during-render pattern)
   255→  const [prevLen, setPrevLen] = useState(records.length);
   256→  if (prevLen !== records.length) {
   257→    setPrevLen(records.length);
   258→    if (cursor !== null) {
   259→      setCursor(records.length === 0 ? null : Math.min(cursor, records.length - 1));
   260→    }
   261→  }
   262→
   263→  // follow the cursor with the scroll container
   264→  useEffect(() => {
   265→    if (cursor === null) return;
   266→    rowRefs.current.get(cursor)?.scrollIntoView({ block: "nearest" });
   267→  }, [cursor]);
   268→
   269→  const addRecord = useMutation({
   270→    mutationFn: () => createRecord(systemId, draft),
   271→    onSuccess: () => {
   272→      invalidateAll();
   273→      setAddingRecord(false);
   274→      setDraft({});
   275→      toast.success(t.recs.toastAddOk, {
   276→        description: t.recs.toastAddDesc,
   277→      });
   278→    },
   279→    onError: (err: Error) =>
   280→      toast.error(t.recs.toastAddErr, { description: err.message }),
   281→  });
   282→
   283→  const removeRecord = useMutation({
   284→    mutationFn: (id: string) => deleteRecord(id),
   285→    onSuccess: (_data, removedId) => {
   286→      invalidateAll();
   287→      setSelected((prev) => {
   288→        const next = new Set(prev);
   289→        next.delete(removedId);
   290→        return next;
   291→      });
   292→      toast(t.recs.toastDelOk, { description: t.recs.toastDelDesc });
   293→    },
   294→    onError: (err: Error) =>
   295→      toast.error(t.recs.toastDelErr, { description: err.message }),
   296→  });
   297→
   298→  const editRecord = useMutation({
   299→    mutationFn: () => updateRecord(editingId!, editDraft),
   300→    onSuccess: () => {
   301→      invalidateAll();
   302→      setEditingId(null);
   303→      setEditDraft({});
   304→      toast.success(t.recs.toastUpdOk, {
   305→        description: t.recs.toastUpdDesc,
   306→      });
   307→    },
   308→    onError: (err: Error) =>
   309→      toast.error(t.recs.toastUpdErr, { description: err.message }),
   310→  });
   311→
   312→  const bulkDelete = useMutation({
   313→    mutationFn: () => bulkDeleteRecords([...selected]),
   314→    onSuccess: (res) => {
   315→      invalidateAll();
   316→      setSelected(new Set());
   317→      setConfirmBulk(false);
   318→      toast.success(t.recs.toastBulkOk(res.deleted), {
   319→        description: t.recs.toastBulkDesc,
   320→      });
   321→    },
   322→    onError: (err: Error) => {
   323→      setConfirmBulk(false);
   324→      toast.error(t.recs.toastBulkErr, { description: err.message });
   325→    },
   326→  });
   327→
   328→  const toggleRow = (id: string) =>
   329→    setSelected((prev) => {
   330→      const next = new Set(prev);
   331→      if (next.has(id)) next.delete(id);
   332→      else next.add(id);
   333→      return next;
   334→    });
   335→
   336→  const allSelected = records.length > 0 && selected.size === records.length;
   337→  const someSelected = selected.size > 0 && !allSelected;
   338→
   339→  const toggleAll = () =>
   340→    setSelected(allSelected ? new Set() : new Set(records.map((r) => r.id)));
   341→
   342→  const selectedRecords = records.filter((r) => selected.has(r.id));
   343→
   344→  /** power-user keyboard model: j/k move, x toggles, e edits, esc dismisses */
   345→  const onTableKeyDown = (e: React.KeyboardEvent) => {
   346→    const target = e.target as HTMLElement;
   347→    // let form fields and open menus consume their own keys
   348→    if (target.closest(
   349→      "input, textarea, select, [role='combobox'], [role='listbox'], [role='option'], [role='menu'], [data-radix-popper-content-wrapper]"
   350→    ))
   351→      return;
   352→    if (records.length === 0) return;
   353→    const key = e.key;
   354→    if (key === "j" || key === "J" || key === "ArrowDown") {
   355→      e.preventDefault();
   356→      setCursor((c) => (c === null ? 0 : Math.min(c + 1, records.length - 1)));
   357→    } else if (key === "k" || key === "K" || key === "ArrowUp") {
   358→      e.preventDefault();
   359→      setCursor((c) => (c === null ? 0 : Math.max(c - 1, 0)));
   360→    } else if (key === "x" || key === "X") {
   361→      if (cursor === null || isArchived) return;
   362→      e.preventDefault();
   363→      toggleRow(records[cursor].id);
   364→    } else if (key === "e" || key === "E") {
   365→      if (cursor === null || isArchived) return;
   366→      e.preventDefault();
   367→      const rec = records[cursor];
   368→      setAddingRecord(false);
   369→      setEditingId(rec.id);
   370→      setEditDraft({ ...parseRecordData(rec.data) });
   371→    } else if (key === "Escape") {
   372→      setCursor(null);
   373→    }
   374→  };
   375→
   376→  return (
   377→    <section className="flex min-w-0 flex-col gap-2.5">
   378→      <div className="flex items-center justify-between gap-2">
   379→        <MonoLabel>{t.recs.header(records.length)}</MonoLabel>
   380→        <div className="flex items-center gap-1.5">
   381→          <Button
   382→            size="sm"
   383→            variant="ghost"
   384→            onClick={() => exportSystemCsv(systemName, columns, records)}
   385→            disabled={records.length === 0}
   386→            className="h-8 gap-1.5 rounded-sm border border-border font-mono text-[10px] uppercase tracking-[0.12em] text-muted-foreground hover:bg-secondary hover:text-foreground disabled:opacity-40"
   387→          >
   388→            <Download className="size-3" /> CSV
   389→          </Button>
   390→          <Button
   391→            size="sm"
   392→            onClick={() => {
   393→              setEditingId(null);
   394→              setAddingRecord((v) => !v);
   395→            }}
   396→            disabled={isArchived}
   397→            className="h-8 gap-1.5 rounded-sm bg-primary font-mono text-[10px] uppercase tracking-[0.12em] text-primary-foreground hover:bg-gold-pale"
   398→          >
   399→            {addingRecord ? (
   400→              <>
   401→                <X className="size-3" /> {t.recs.cancel}
   402→              </>
   403→            ) : (
   404→              <>
   405→                <Plus className="size-3" /> {t.recs.add}
   406→              </>
   407→            )}
   408→          </Button>
   409→        </div>
   410→      </div>
   411→
   412→      {/* bulk selection toolbar */}
   413→      <AnimatePresence>
   414→        {selected.size > 0 && (
   415→          <motion.div
   416→            initial={{ opacity: 0, height: 0 }}
   417→            animate={{ opacity: 1, height: "auto" }}
   418→            exit={{ opacity: 0, height: 0 }}
   419→            className="overflow-hidden"
   420→          >
   421→            <div className="flex flex-wrap items-center gap-2 rounded-lg border border-gold/35 bg-gold/[0.05] px-3 py-2">
   422→              <span className="font-mono text-[10px] uppercase tracking-[0.14em] text-gold">
   423→                {t.recs.selected(selected.size)}
   424→              </span>
   425→              <div className="ms-auto flex items-center gap-1.5">
   426→                <Button
   427→                  size="sm"
   428→                  variant="ghost"
   429→                  onClick={() =>
   430→                    exportSystemCsv(systemName, columns, selectedRecords, "selection")
   431→                  }
   432→                  className="h-7 gap-1.5 rounded-sm px-2 font-mono text-[10px] uppercase tracking-[0.12em] text-muted-foreground hover:bg-secondary hover:text-foreground"
   433→                >
   434→                  <Download className="size-3" /> {t.recs.export}
   435→                </Button>
   436→                <Button
   437→                  size="sm"
   438→                  variant="ghost"
   439→                  onClick={() => setSelected(new Set())}
   440→                  className="h-7 rounded-sm px-2 font-mono text-[10px] uppercase tracking-[0.12em] text-muted-foreground hover:bg-secondary hover:text-foreground"
   441→                >
   442→                  {t.recs.clearSel}
   443→                </Button>
   444→                <Button
   445→                  size="sm"
   446→                  variant="ghost"
   447→                  onClick={() => setConfirmBulk(true)}
   448→                  disabled={isArchived || bulkDelete.isPending}
   449→                  className="h-7 gap-1.5 rounded-sm border border-destructive/40 px-2 font-mono text-[10px] uppercase tracking-[0.12em] text-destructive hover:bg-destructive/10"
   450→                >
   451→                  {bulkDelete.isPending ? (
   452→                    <Loader2 className="size-3 animate-spin" />
   453→                  ) : (
   454→                    <Trash2 className="size-3" />
   455→                  )}
   456→                  {t.recs.deleteSel}
   457→                </Button>
   458→              </div>
   459→            </div>
   460→          </motion.div>
   461→        )}
   462→      </AnimatePresence>
   463→
   464→      {/* inline add form */}
   465→      <AnimatePresence>
   466→        {addingRecord && (
   467→          <motion.form
   468→            initial={{ opacity: 0, height: 0 }}
   469→            animate={{ opacity: 1, height: "auto" }}
   470→            exit={{ opacity: 0, height: 0 }}
   471→            onSubmit={(e) => {
   472→              e.preventDefault();
   473→              addRecord.mutate();
   474→            }}
   475→            className="overflow-hidden"
   476→          >
   477→            <div className="grid gap-3 rounded-lg border border-gold/25 bg-gold/[0.04] p-4 sm:grid-cols-2">
   478→              {columns.map((f) => (
   479→                <div key={f.key} className="flex flex-col gap-1.5">
   480→                  <Label className="font-mono text-[9px] uppercase tracking-[0.14em] text-muted-foreground">
   481→                    {f.label}
   482→                  </Label>
   483→                  <DynamicFieldInput
   484→                    field={f}
   485→                    value={draft[f.key]}
   486→                    onChange={(v) =>
   487→                      setDraft((d) => ({ ...d, [f.key]: v }))
   488→                    }
   489→                  />
   490→                </div>
   491→              ))}
   492→              {columns.length === 0 && (
   493→                <p className="text-sm text-muted-foreground">
   494→                  {t.recs.noFields}
   495→                </p>
   496→              )}
   497→              <div className="sm:col-span-2">
   498→                <Button
   499→                  type="submit"
   500→                  disabled={
   501→                    addRecord.isPending || columns.length === 0
   502→                  }
   503→                  className="h-9 w-full bg-primary font-mono text-[10px] uppercase tracking-[0.14em] text-primary-foreground hover:bg-gold-pale sm:w-48"
   504→                >
   505→                  {addRecord.isPending ? (
   506→                    <>
   507→                      <Loader2 className="animate-spin" /> {t.recs.saving}
   508→                    </>
   509→                  ) : (
   510→                    t.recs.saveRecord
   511→                  )}
   512→                </Button>
   513→              </div>
   514→            </div>
   515→          </motion.form>
   516→        )}
   517→      </AnimatePresence>
   518→
   519→      {/* inline edit form */}
   520→      <AnimatePresence>
   521→        {editingId && !addingRecord && (
   522→          <motion.form
   523→            key={editingId}
   524→            initial={{ opacity: 0, height: 0 }}
   525→            animate={{ opacity: 1, height: "auto" }}
   526→            exit={{ opacity: 0, height: 0 }}
   527→            onSubmit={(e) => {
   528→              e.preventDefault();
   529→              editRecord.mutate();
   530→            }}
   531→            className="overflow-hidden"
   532→          >
   533→            <div className="grid gap-3 rounded-lg border border-gold/40 bg-gold/[0.06] p-4 sm:grid-cols-2">
   534→              <div className="flex items-center gap-2 sm:col-span-2">
   535→                <Pencil className="size-3.5 text-gold" />
   536→                <MonoLabel gold className="text-[9px]">
   537→                  {t.recs.editing}
   538→                </MonoLabel>
   539→              </div>
   540→              {columns.map((f) => (
   541→                <div key={f.key} className="flex flex-col gap-1.5">
   542→                  <Label className="font-mono text-[9px] uppercase tracking-[0.14em] text-muted-foreground">
   543→                    {f.label}
   544→                  </Label>
   545→                  <DynamicFieldInput
   546→                    field={f}
   547→                    value={editDraft[f.key]}
   548→                    onChange={(v) =>
   549→                      setEditDraft((d) => ({ ...d, [f.key]: v }))
   550→                    }
   551→                  />
   552→                </div>
   553→              ))}
   554→              <div className="flex gap-2 sm:col-span-2">
   555→                <Button
   556→                  type="submit"
   557→                  disabled={editRecord.isPending || columns.length === 0}
   558→                  className="h-9 flex-1 bg-primary font-mono text-[10px] uppercase tracking-[0.14em] text-primary-foreground hover:bg-gold-pale sm:flex-none sm:px-6"
   559→                >
   560→                  {editRecord.isPending ? (
   561→                    <>
   562→                      <Loader2 className="animate-spin" /> {t.recs.saving}
   563→                    </>
   564→                  ) : (
   565→                    t.recs.saveChanges
   566→                  )}
   567→                </Button>
   568→                <Button
   569→                  type="button"
   570→                  variant="ghost"
   571→                  onClick={() => {
   572→                    setEditingId(null);
   573→                    setEditDraft({});
   574→                  }}
   575→                  className="h-9 border border-border font-mono text-[10px] uppercase tracking-[0.14em] text-muted-foreground hover:bg-secondary hover:text-foreground"
   576→                >

   578→                </Button>
   579→              </div>
   580→            </div>
   581→          </motion.form>
   582→        )}
   583→      </AnimatePresence>
   584→
   585→      {/* records table */}

   590→          className="py-8"

   593→        <>
   594→          <div
   595→            tabIndex={0}
   596→            onKeyDown={onTableKeyDown}
   597→            aria-label={t.recs.header(records.length)}
   598→            className="min-w-0 overflow-hidden rounded-lg border border-border transition-shadow focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-gold/50"

   641→                          "bg-gold/[0.06] hover:bg-gold/[0.09]",
   642→                        isCursor &&
   643→                          !isSelected &&
   644→                          !isEditing &&
   645→                          "bg-gold/[0.045] hover:bg-gold/[0.07]",
   646→                        // gold inline-start rail marks the keyboard cursor
   647→                        isCursor &&
   648→                          "shadow-[inset_2px_0_0_0_#E8B44A] rtl:shadow-[inset_-2px_0_0_0_#E8B44A]"

   667→                      ))}
   668→                      <TableCell>
   669→                        <DropdownMenu>
   670→                          <DropdownMenuTrigger asChild>
   671→                            <button
   672→                              className="flex size-8 items-center justify-center rounded-sm text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/60"
   673→                              aria-label={t.recs.recActions}

   675→                              <MoreHorizontal className="size-4" />

   683→                              onClick={() => {
   684→                                setAddingRecord(false);
   685→                                setEditingId(rec.id);
   686→                                setEditDraft({ ...data });

   699→                          </DropdownMenuContent>
   700→                        </DropdownMenu>
   701→                      </TableCell>
   702→                    </TableRow>
   703→                  );
   704→                })}
   705→              </TableBody>
   706→            </Table>
   707→          </div>
   708→          {/* keyboard hints — OPS-DECK power-user affordance */}
   709→          <div
   710→            className="flex flex-wrap items-center gap-x-3 gap-y-1 px-0.5 font-mono text-[9px] uppercase tracking-[0.12em] text-muted-foreground/60"
   711→            aria-hidden
   712→          >
   713→            <span className="flex items-center gap-1">
   714→              <Kbd>J</Kbd>
   715→              <Kbd>K</Kbd> {t.recs.kbdNav}
   716→            </span>
   717→            <span className="flex items-center gap-1">
   718→              <Kbd>X</Kbd> {t.recs.kbdSelect}
   719→            </span>
   720→            <span className="flex items-center gap-1">
   721→              <Kbd>E</Kbd> {t.recs.kbdEdit}
   722→            </span>
   723→            <span className="flex items-center gap-1">
   724→              <Kbd>Esc</Kbd> {t.recs.kbdClear}
   725→            </span>
   726→          </div>
   727→        </>

   729→
   730→      {/* bulk delete confirm */}
   731→      <AlertDialog open={confirmBulk} onOpenChange={setConfirmBulk}>
   732→        <AlertDialogContent className="border-border bg-card">
   733→          <AlertDialogHeader>

   740→          </AlertDialogHeader>

   745→            <AlertDialogAction
   746→              onClick={(e) => {

   750→              className="bg-destructive font-mono text-[11px] uppercase tracking-[0.12em] text-white hover:bg-destructive/90"
